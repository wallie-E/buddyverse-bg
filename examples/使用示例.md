# 微信交换功能 - 实际使用示例

## 场景 1: 在帖子详情页添加交换按钮

### 需求
用户浏览帖子时，可以点击作者头像旁边的"交换微信"按钮。

### 实现代码

```jsx
import React, { useState } from 'react';
import WechatExchangeModal from './components/WechatExchangeModal';

function PostDetail({ post }) {
  const [showExchangeModal, setShowExchangeModal] = useState(false);
  const currentUserId = getCurrentUserId(); // 从全局状态或 context 获取

  // 不能与自己交换
  const canExchange = currentUserId !== post.user_id;

  return (
    <div className="post-detail">
      {/* 帖子内容 */}
      <div className="post-content">{post.content}</div>

      {/* 作者信息 */}
      <div className="post-author">
        <img src={post.author.avatar} alt={post.author.nickname} />
        <span>{post.author.nickname}</span>
        
        {/* 交换微信按钮 */}
        {canExchange && (
          <button 
            className="exchange-btn"
            onClick={() => setShowExchangeModal(true)}
          >
            💬 交换微信
          </button>
        )}
      </div>

      {/* 交换弹窗 */}
      <WechatExchangeModal
        visible={showExchangeModal}
        targetUserId={post.user_id}
        targetUserName={post.author.nickname}
        onClose={() => setShowExchangeModal(false)}
        onSuccess={() => {
          setShowExchangeModal(false);
          alert('操作成功！');
        }}
      />
    </div>
  );
}
```

## 场景 2: 在用户资料页展示交换状态

### 需求
在用户资料页显示与该用户的交换状态，并提供操作按钮。

### 实现代码

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';
import WechatExchangeModal from './components/WechatExchangeModal';

function UserProfile({ userId }) {
  const [userInfo, setUserInfo] = useState(null);
  const [exchangeStatus, setExchangeStatus] = useState(null);
  const [showExchangeModal, setShowExchangeModal] = useState(false);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchUserAndExchangeInfo();
  }, [userId]);

  const fetchUserAndExchangeInfo = async () => {
    try {
      setLoading(true);
      
      // 获取用户信息
      const userRes = await axios.get(`/api/users/${userId}`, {
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      });
      setUserInfo(userRes.data.data);

      // 获取交换状态
      const exchangeRes = await axios.get('/api/wechat-exchange/info', {
        params: { targetUserId: userId },
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      });
      setExchangeStatus(exchangeRes.data.data);
    } catch (err) {
      console.error('获取信息失败:', err);
    } finally {
      setLoading(false);
    }
  };

  const getExchangeStatusText = () => {
    if (!exchangeStatus || !exchangeStatus.exists) {
      return '未交换';
    }

    switch (exchangeStatus.status) {
      case 0:
        if (exchangeStatus.waitingFor === 'other') {
          return '等待对方确认';
        } else if (exchangeStatus.waitingFor === 'me') {
          return '等待您确认';
        }
        return '交换中';
      case 1:
        return '已交换';
      case 2:
        return '已拒绝';
      case 3:
        return '已过期';
      default:
        return '未知状态';
    }
  };

  const getButtonText = () => {
    if (!exchangeStatus || !exchangeStatus.exists) {
      return '交换微信';
    }

    switch (exchangeStatus.status) {
      case 0:
        if (exchangeStatus.waitingFor === 'other') {
          return '重新发送';
        } else if (exchangeStatus.waitingFor === 'me') {
          return '确认交换';
        }
        return '查看详情';
      case 1:
        return '查看微信号';
      default:
        return '交换微信';
    }
  };

  if (loading) {
    return <div>加载中...</div>;
  }

  return (
    <div className="user-profile">
      {/* 用户头像和基本信息 */}
      <div className="profile-header">
        <img src={userInfo.avatar} alt={userInfo.nickname} />
        <div className="profile-info">
          <h2>{userInfo.nickname}</h2>
          <p>{userInfo.signature}</p>
        </div>
      </div>

      {/* 交换状态卡片 */}
      <div className="exchange-status-card">
        <div className="status-header">
          <span className="status-label">微信交换状态</span>
          <span className={`status-badge status-${exchangeStatus?.status || 'none'}`}>
            {getExchangeStatusText()}
          </span>
        </div>

        {/* 已完成交换，显示微信号 */}
        {exchangeStatus?.status === 1 && (
          <div className="wechat-info">
            <p><strong>对方微信号：</strong></p>
            <div className="wechat-id-box">
              <span className="wechat-id">{exchangeStatus.otherWechat}</span>
              <button 
                onClick={() => {
                  navigator.clipboard.writeText(exchangeStatus.otherWechat);
                  alert('已复制到剪贴板');
                }}
              >
                📋 复制
              </button>
            </div>
          </div>
        )}

        {/* 操作按钮 */}
        <button 
          className="action-btn"
          onClick={() => setShowExchangeModal(true)}
        >
          {getButtonText()}
        </button>
      </div>

      {/* 交换弹窗 */}
      <WechatExchangeModal
        visible={showExchangeModal}
        targetUserId={userId}
        targetUserName={userInfo.nickname}
        onClose={() => setShowExchangeModal(false)}
        onSuccess={() => {
          setShowExchangeModal(false);
          fetchUserAndExchangeInfo(); // 刷新状态
        }}
      />
    </div>
  );
}
```

## 场景 3: 通知中心处理交换请求

### 需求
用户在通知中心看到交换请求，点击后可以直接确认或拒绝。

### 实现代码

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';
import WechatExchangeModal from './components/WechatExchangeModal';

function NotificationCenter() {
  const [notifications, setNotifications] = useState([]);
  const [unreadCount, setUnreadCount] = useState(0);
  const [showExchangeModal, setShowExchangeModal] = useState(false);
  const [selectedExchange, setSelectedExchange] = useState(null);

  useEffect(() => {
    fetchNotifications();
    fetchUnreadCount();
  }, []);

  const fetchNotifications = async () => {
    try {
      const res = await axios.get('/api/notifications', {
        params: { page: 1, limit: 50 },
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      });
      setNotifications(res.data.data.list);
    } catch (err) {
      console.error('获取通知失败:', err);
    }
  };

  const fetchUnreadCount = async () => {
    try {
      const res = await axios.get('/api/notifications/unread-count', {
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      });
      setUnreadCount(res.data.data.count);
    } catch (err) {
      console.error('获取未读数失败:', err);
    }
  };

  const handleNotificationClick = async (notification) => {
    // 标记已读
    if (!notification.is_read) {
      await axios.put(
        `/api/notifications/${notification.id}/read`,
        {},
        { headers: { Authorization: `Bearer ${localStorage.getItem('token')}` } }
      );
    }

    // 处理微信交换通知
    if (notification.type === 'wechat_exchange_request' || 
        notification.type === 'wechat_exchange_confirmed') {
      
      // 从通知内容中提取用户信息
      const userName = extractUserName(notification.content);
      
      setSelectedExchange({
        userId: notification.related_user_id || notification.related_id,
        userName
      });
      setShowExchangeModal(true);
    }

    fetchNotifications();
    fetchUnreadCount();
  };

  const extractUserName = (content) => {
    const match = content.match(/^(.+?)\s/);
    return match ? match[1] : '用户';
  };

  const getNotificationIcon = (type) => {
    const icons = {
      wechat_exchange_request: '💬',
      wechat_exchange_confirmed: '✅',
      comment: '💭',
      reply: '↩️',
      system: '📢'
    };
    return icons[type] || '🔔';
  };

  return (
    <div className="notification-center">
      <div className="notification-header">
        <h2>通知中心</h2>
        {unreadCount > 0 && (
          <span className="unread-badge">{unreadCount}</span>
        )}
      </div>

      <div className="notification-list">
        {notifications.map(notification => (
          <div
            key={notification.id}
            className={`notification-item ${!notification.is_read ? 'unread' : ''}`}
            onClick={() => handleNotificationClick(notification)}
          >
            <div className="notification-icon">
              {getNotificationIcon(notification.type)}
            </div>
            <div className="notification-content">
              <div className="notification-title">{notification.title}</div>
              <div className="notification-text">{notification.content}</div>
              <div className="notification-time">
                {new Date(notification.created_at).toLocaleString()}
              </div>
            </div>
            {!notification.is_read && <div className="unread-dot"></div>}
          </div>
        ))}
      </div>

      {/* 交换弹窗 */}
      {selectedExchange && (
        <WechatExchangeModal
          visible={showExchangeModal}
          targetUserId={selectedExchange.userId}
          targetUserName={selectedExchange.userName}
          onClose={() => {
            setShowExchangeModal(false);
            setSelectedExchange(null);
          }}
          onSuccess={() => {
            setShowExchangeModal(false);
            setSelectedExchange(null);
            fetchNotifications();
          }}
        />
      )}
    </div>
  );
}
```

## 场景 4: 我的交换记录页面

### 需求
展示用户所有的微信交换记录，可以筛选状态。

### 实现代码

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';

function MyExchanges() {
  const [exchanges, setExchanges] = useState([]);
  const [filter, setFilter] = useState('all'); // all, pending, completed
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchExchanges();
  }, [filter]);

  const fetchExchanges = async () => {
    try {
      setLoading(true);
      const params = { page: 1, limit: 50 };
      if (filter === 'pending') params.status = 0;
      if (filter === 'completed') params.status = 1;

      const res = await axios.get('/api/wechat-exchange/my-exchanges', {
        params,
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      });
      setExchanges(res.data.data.list);
    } catch (err) {
      console.error('获取交换记录失败:', err);
    } finally {
      setLoading(false);
    }
  };

  const getStatusText = (status) => {
    const statusMap = {
      0: '待确认',
      1: '已完成',
      2: '已拒绝',
      3: '已过期'
    };
    return statusMap[status] || '未知';
  };

  const copyWechatId = (wechatId) => {
    navigator.clipboard.writeText(wechatId);
    alert('已复制到剪贴板');
  };

  return (
    <div className="my-exchanges">
      <h2>我的微信交换</h2>

      {/* 筛选按钮 */}
      <div className="filter-buttons">
        <button 
          className={filter === 'all' ? 'active' : ''}
          onClick={() => setFilter('all')}
        >
          全部
        </button>
        <button 
          className={filter === 'pending' ? 'active' : ''}
          onClick={() => setFilter('pending')}
        >
          待确认
        </button>
        <button 
          className={filter === 'completed' ? 'active' : ''}
          onClick={() => setFilter('completed')}
        >
          已完成
        </button>
      </div>

      {/* 交换记录列表 */}
      {loading ? (
        <div className="loading">加载中...</div>
      ) : exchanges.length === 0 ? (
        <div className="empty-state">暂无交换记录</div>
      ) : (
        <div className="exchange-list">
          {exchanges.map(exchange => (
            <div key={exchange.id} className="exchange-card">
              <div className="exchange-header">
                <div className="user-info">
                  <img src={exchange.otherAvatar} alt={exchange.otherNickname} />
                  <span>{exchange.otherNickname}</span>
                </div>
                <span className={`status-badge status-${exchange.status}`}>
                  {getStatusText(exchange.status)}
                </span>
              </div>

              {exchange.status === 1 && (
                <div className="wechat-info">
                  <div className="wechat-row">
                    <span className="label">我的微信：</span>
                    <span className="wechat-id">{exchange.myWechat}</span>
                  </div>
                  <div className="wechat-row">
                    <span className="label">对方微信：</span>
                    <span className="wechat-id">{exchange.otherWechat}</span>
                    <button onClick={() => copyWechatId(exchange.otherWechat)}>
                      复制
                    </button>
                  </div>
                </div>
              )}

              <div className="exchange-time">
                {new Date(exchange.createdAt).toLocaleString()}
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

## 完整的 API 调用示例

### 使用 Axios 封装

```javascript
// api/wechatExchange.js
import axios from 'axios';

const API_BASE_URL = 'http://localhost:8080/api';

const getAuthHeaders = () => ({
  Authorization: `Bearer ${localStorage.getItem('token')}`
});

export const wechatExchangeAPI = {
  // 获取交换信息
  getInfo: async (targetUserId) => {
    const response = await axios.get(`${API_BASE_URL}/wechat-exchange/info`, {
      params: { targetUserId },
      headers: getAuthHeaders()
    });
    return response.data;
  },

  // 发起交换请求
  request: async (targetUserId, wechatId, isUpdate = false) => {
    const response = await axios.post(
      `${API_BASE_URL}/wechat-exchange/request`,
      { targetUserId, wechatId, isUpdate },
      { headers: getAuthHeaders() }
    );
    return response.data;
  },

  // 确认交换
  confirm: async (exchangeNo, wechatId, action) => {
    const response = await axios.post(
      `${API_BASE_URL}/wechat-exchange/confirm`,
      { exchangeNo, wechatId, action },
      { headers: getAuthHeaders() }
    );
    return response.data;
  },

  // 获取交换记录
  getMyExchanges: async (page = 1, limit = 20, status = null) => {
    const params = { page, limit };
    if (status !== null) params.status = status;
    
    const response = await axios.get(`${API_BASE_URL}/wechat-exchange/my-exchanges`, {
      params,
      headers: getAuthHeaders()
    });
    return response.data;
  }
};
```

## 最佳实践

### 1. 错误处理

```javascript
const handleExchangeWithErrorHandling = async () => {
  try {
    const result = await wechatExchangeAPI.request(targetUserId, wechatId);
    
    if (result.success) {
      // 成功处理
      showSuccessMessage(result.message);
    }
  } catch (error) {
    // 错误处理
    if (error.response) {
      // 服务器返回错误
      const { status, data } = error.response;
      
      switch (status) {
        case 400:
          showError(data.message || '请求参数错误');
          break;
        case 401:
          showError('请先登录');
          redirectToLogin();
          break;
        case 403:
          showError('没有权限');
          break;
        case 404:
          showError('用户不存在');
          break;
        default:
          showError('操作失败，请稍后重试');
      }
    } else {
      // 网络错误
      showError('网络连接失败');
    }
  }
};
```
